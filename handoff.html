Three prompts to paste into Lovable, in order. They add two standalone landing pages at /wilcox and /wilcox-lineup without touching anything already on the site. Everything the prompts need is included in them, so there is nothing to reverse engineer from the preview.

Read this first

The design is finished. Do not let the AI rebuild it. The prompts hand over tested CSS and say so explicitly. If Lovable offers to redo it in Tailwind or with shadcn components, decline. That is the one thing most likely to change how these look.

Every CSS rule is scoped to a single .bomal-lp wrapper, so none of it can reach the rest of bomal.org. That scoping has been tested against a hostile host page. Do not remove it.

These pages intentionally have no header, footer or navigation. They are ad destinations, and every link out of them is a way to lose the registration. The only outbound links go to register.bomal.org.

Route names are a suggestion. Change /wilcox and /wilcox-lineup to whatever you prefer in prompt 1, just keep them consistent in prompts 2 and 3.

var SPEAKERS = {

/**
 * Wire up one landing page. Call it from useEffect with the wrapper element.
 * Returns a cleanup function that clears the countdown interval.
 */
export function initLandingPage(root) {
  if (!root) return function () {};
  var timer = null;

    var params = new URLSearchParams(window.location.search);
    var assetBase = root.getAttribute('data-asset-base') || 'brand_assets/';

    /* ---------- 1. Speaker message-match ----------
       Two separate questions. Which speaker does the page show, and did the
       visitor actually arrive from that speaker's ad? A page can carry a default
       speaker (data-default-speaker) so it is never generic, but the
       "you followed X here" copy must only appear for a real ad click. */
    var has = function (k) { return !!k && Object.prototype.hasOwnProperty.call(SPEAKERS, k); };

    var urlKey = (params.get('s') || params.get('speaker') || '').toLowerCase().trim();
    var defKey = (root.getAttribute('data-default-speaker') || '').toLowerCase().trim();

    /* A dedicated page is the destination of one speaker's ad, so every visitor
       did arrive from that ad whether or not the URL still carries ?s=. On a
       shared page, only an explicit ?s= counts. */
    var dedicated = root.hasAttribute('data-speaker-dedicated');

    var key = has(urlKey) ? urlKey : (has(defKey) ? defKey : '');
    var fromAd = has(urlKey) || (dedicated && has(defKey));
    var speaker = key ? SPEAKERS[key] : null;

    if (speaker) {
      root.querySelectorAll('[data-sp="name"]').forEach(function (el) { el.textContent = speaker.name; });
      root.querySelectorAll('[data-sp="role"]').forEach(function (el) { el.textContent = speaker.role; });
      root.querySelectorAll('[data-sp="img"]').forEach(function (el) {
        el.src = assetBase + 'speakers/' + speaker.img;
        el.alt = speaker.name;
      });
      /* Whoever is in the hero must not also appear in the lineup grid below,
         default speaker included. */
      root.querySelectorAll('[data-lineup="' + key + '"]').forEach(function (el) { el.hidden = true; });
    }

    /* Only a real ad click earns the "you followed X here" copy. */
    if (fromAd) {
      root.querySelectorAll('[data-sp-only]').forEach(function (el) { el.hidden = false; });
      root.querySelectorAll('[data-sp-none]').forEach(function (el) { el.hidden = true; });
    }

    /* ---------- 2. Urgency ---------- */
    var deadline = new Date(EARLY_BIRD_END);
    var live = !isNaN(deadline) && deadline.getTime() > Date.now();

    function remaining() {
      var ms = deadline.getTime() - Date.now();
      if (ms <= 0) return null;
      var d = Math.floor(ms / 864e5);
      var h = Math.floor((ms % 864e5) / 36e5);
      if (d >= 1) return d + (d === 1 ? ' day ' : ' days ') + h + (h === 1 ? ' hour' : ' hours');
      var m = Math.floor((ms % 36e5) / 6e4);
      return h + (h === 1 ? ' hour ' : ' hours ') + m + (m === 1 ? ' minute' : ' minutes');
    }

    function paintUrgency() {
      var left = live ? remaining() : null;
      if (left) {
        root.querySelectorAll('[data-urgency]').forEach(function (el) {
          el.innerHTML = 'Early-bird pricing ends in <b>' + left + '</b>. Register now for <b>$25</b> instead of $30.';
        });
        root.querySelectorAll('[data-price-now]').forEach(function (el) { el.textContent = '$25'; });
        root.querySelectorAll('[data-price-was]').forEach(function (el) { el.hidden = false; });
        root.querySelectorAll('[data-price-note]').forEach(function (el) {
          el.textContent = 'Early-bird price, through October 10.';
        });
      } else {
        live = false;
        root.querySelectorAll('[data-urgency]').forEach(function (el) {
          el.innerHTML = '<b>November 6 and 7</b> at the Provo Marriott. Banquet and box lunch tickets are available while they last.';
        });
        root.querySelectorAll('[data-price-now]').forEach(function (el) { el.textContent = '$30'; });
        root.querySelectorAll('[data-price-was]').forEach(function (el) { el.hidden = true; });
        root.querySelectorAll('[data-price-note]').forEach(function (el) {
          el.textContent = 'Full registration, both days.';
        });
      }
    }
    paintUrgency();
    if (live) timer = setInterval(paintUrgency, 60000);

    /* ---------- 3. UTM passthrough ---------- */
    var carry = ['utm_source','utm_medium','utm_campaign','utm_content','utm_term','fbclid','gclid','ttclid'];
    var pass = new URLSearchParams();
    carry.forEach(function (k) { if (params.get(k)) pass.set(k, params.get(k)); });
    if (fromAd && key) pass.set('utm_content', pass.get('utm_content') || key);

    var qs = pass.toString();
    root.querySelectorAll('[data-register]').forEach(function (el) {
      el.setAttribute('href', REGISTER_URL + (qs ? '?' + qs : ''));
    });

    /* ---------- FAQ disclosure, where a page has one ---------- */
    root.querySelectorAll('[data-faq] button').forEach(function (btn) {
      btn.addEventListener('click', function () {
        var open = btn.getAttribute('aria-expanded') === 'true';
        btn.setAttribute('aria-expanded', String(!open));
        var panel = document.getElementById(btn.getAttribute('aria-controls'));
        if (panel) panel.hidden = open;
      });
    });
  return function () { if (timer) clearInterval(timer); };
}

Worth knowing before you publish

The price changes itself. One date near the top of bomal-lp.js, EARLY_BIRD_END, drives the countdown bar and every price on both pages. Until it passes they read $25 with $30 struck through; after it they flip to $30 and the event date on their own. Change that one line if the deadline moves.

Ad tracking is already wired. Every button carries any utm_*, fbclid or gclid through to register.bomal.org, and falls back to utm_content=wilcox when the ad platform sends nothing, so these registrations stay countable.

These pages should be indexable once live. The preview they came from is blocked by a robots.txt that is not part of the handoff, so there is nothing to undo.

A correction worth carrying across. register.bomal.org currently shows the Friday banquet and the Saturday box lunch as sold out. They are not. These pages say they are available while tickets last, which means the two sites disagree until the registration page is updated.
